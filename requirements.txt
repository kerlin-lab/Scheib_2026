# --- Core scientific stack -------------------------------------------------
# NOTE: numpy MUST stay on the 1.x line. js_manuscript_final.py installs a
# compatibility shim (sys.modules.setdefault('numpy._core', numpy.core)) so that
# pickles written under numpy 2.x load under numpy 1.x. Installing numpy>=2
# changes the pickle module paths and defeats that shim.
numpy==1.24.4
scipy==1.10.1
pandas==1.4.2

# --- Statistics / modeling -------------------------------------------------
statsmodels==0.14.0
scikit-learn==1.1.1

# --- Plotting --------------------------------------------------------------

# --- Imaging IO ------------------------------------------------------------
# imagecodecs backs tifffile's compressed-TIFF readers (Fig1 behavior frames,
# Fig2 z-stack projections).
tifffile==2022.5.4
imagecodecs==2021.8.26

# --- Progress bars ---------------------------------------------------------
tqdm==4.64.1

# --- Parallel compute (Fig4 only) ------------------------------------------
# Fig4.ipynb builds a dask.distributed LocalCluster for the heavy SMO passes.
# dask and distributed are released in lockstep and distributed pins its dask
# exactly (distributed 2022.2.1 requires dask==2022.02.1), so these two must
# move together. The reference sled_env happens to hold dask 2021.12.0 next to
# distributed 2022.2.1 -- an inconsistent combination that pip refuses to
# resolve; 2022.02.1 is the matching dask release for that distributed.
dask==2022.02.1
distributed==2022.2.1

# --- Jupyter runtime -------------------------------------------------------
# ipympl backs the `%matplotlib widget` cell in Fig1; pinned for notebook 6.x
# compatibility.
ipython==8.3.0
ipykernel==6.9.1
notebook==6.4.11
ipympl==0.9.3
# notebook 6.4.x declares jupyter-client>=5.3.4 with no upper bound, but its
# kernel-management calls break against jupyter-client 8.x. These are the
# versions the working reference environment holds -- pin them so a fresh
# resolve does not pull 8.x and leave you with a notebook that cannot start a
# kernel.
jupyter-client==7.4.9
jupyter-core==5.8.1
